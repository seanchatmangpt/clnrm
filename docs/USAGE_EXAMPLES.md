## Span Builders

### Basic Span Creation

```rust
use clnrm_core::telemetry::generated::spans::TestExecutionSpan;

fn execute_test(name: &str, image: &str) -> Result<()> {
    // Required attributes as constructor parameters
    // Compiler enforces you provide them!
    let span = TestExecutionSpan::new(name, image);

    // Enter span context
    let _guard = span.enter();

    // Your test logic here
    run_test_steps()?;

    // Span automatically ends when dropped
    Ok(())
}
```

### Span with Optional Attributes

```rust
use clnrm_core::telemetry::generated::spans::TestExecutionSpan;

fn execute_hermetic_test(name: &str, image: &str) -> Result<()> {
    let span = TestExecutionSpan::new(name, image);

    // Optional attributes via chainable setters
    span.set_isolated(true)
        .set_result("pass");

    let _guard = span.enter();

    run_hermetic_test()?;

    Ok(())
}
```

### Nested Spans (Parent-Child)

```rust
use clnrm_core::telemetry::generated::spans::{TestExecutionSpan, StepExecutionSpan};

fn execute_multi_step_test(name: &str, image: &str) -> Result<()> {
    // Parent span
    let test_span = TestExecutionSpan::new(name, image);
    let _test_guard = test_span.enter();

    for (idx, step) in steps.iter().enumerate() {
        // Child span - automatically linked to parent
        let step_span = StepExecutionSpan::new(step.name, idx);
        let _step_guard = step_span.enter();

        execute_step(step)?;

        // step_span ends when _step_guard drops
    }

    // test_span ends when _test_guard drops
    Ok(())
}
```

### Manual Span Control

```rust
use clnrm_core::telemetry::generated::spans::ContainerOperationSpan;

fn start_container(image: &str) -> Result<ContainerId> {
    let span = ContainerOperationSpan::new(image, "start");

    let _guard = span.enter();

    match actually_start_container(image) {
        Ok(id) => {
            span.set_result("success");
            span.set_container_id(&id);
            span.end(); // Explicit end before returning
            Ok(id)
        }
        Err(e) => {
            span.set_result("failure");
            span.set_error_message(&e.to_string());
            span.end();
            Err(e)
        }
    }
}
```

## Metric Recorders

### Recording Histogram Values

```rust
#[cfg(feature = "otel-metrics")]
use clnrm_core::telemetry::generated::metrics::TestExecutionMetric;
use opentelemetry::global;
use std::time::Instant;

#[cfg(feature = "otel-metrics")]
fn measure_test_duration(name: &str, image: &str) -> Result<()> {
    let meter = global::meter("clnrm");
    let metric = TestExecutionMetric::new(&meter);

    let start = Instant::now();

    run_test(name, image)?;

    let duration_ms = start.elapsed().as_secs_f64() * 1000.0;

    // Required attributes enforced in signature
    metric.record(duration_ms, name, "pass");

    Ok(())
}
```

## Mock Testing

### Basic Mock Usage

```rust
#[cfg(test)]
use clnrm_core::telemetry::generated::mocks::MockTestExecutionSpanTrait;
use mockall::predicate::*;

#[test]
fn test_telemetry_calls() {
    let mut mock = MockTestExecutionSpanTrait::new();

    // Expect method called once with specific value
    mock.expect_set_result()
        .with(eq("pass".to_string()))
        .times(1)
        .returning(|_| ());

    // Execute code under test
    mock.set_result("pass".to_string());
}
```

### Mock with Multiple Expectations

```rust
#[cfg(test)]
use clnrm_core::telemetry::generated::mocks::MockTestExecutionSpanTrait;
use mockall::predicate::*;

#[test]
fn test_complete_span_lifecycle() {
    let mut mock = MockTestExecutionSpanTrait::new();

    // Set up multiple expectations
    mock.expect_set_isolated()
        .with(eq(true))
        .times(1)
        .returning(|_| ());

    mock.expect_set_result()
        .with(eq("pass".to_string()))
        .times(1)
        .returning(|_| ());

    mock.expect_end()
        .times(1)
        .returning(|| ());

    // Execute
    mock.set_isolated(true);
    mock.set_result("pass".to_string());
    Box::new(mock).end();
}
```

## Integration Patterns

### Pattern 1: Test Wrapper

```rust
use clnrm_core::telemetry::generated::spans::TestExecutionSpan;

pub struct TelemetryTestRunner {
    name: String,
    image: String,
}

impl TelemetryTestRunner {
    pub fn new(name: impl Into<String>, image: impl Into<String>) -> Self {
        Self {
            name: name.into(),
            image: image.into(),
        }
    }

    pub fn run<F, T>(&self, test_fn: F) -> Result<T>
    where
        F: FnOnce() -> Result<T>,
    {
        let span = TestExecutionSpan::new(&self.name, &self.image);
        let _guard = span.enter();

        let result = test_fn();

        match &result {
            Ok(_) => span.set_result("pass"),
            Err(_) => span.set_result("fail"),
        };

        result
    }
}

// Usage
let runner = TelemetryTestRunner::new("my_test", "alpine:latest");
runner.run(|| {
    // Test code here
    Ok(())
})?;
```

### Pattern 2: Builder Pattern

```rust
use clnrm_core::telemetry::generated::spans::TestExecutionSpan;

pub struct TestBuilder {
    name: String,
    image: String,
    isolated: bool,
    timeout_seconds: Option<u64>,
}

impl TestBuilder {
    pub fn new(name: impl Into<String>, image: impl Into<String>) -> Self {
        Self {
            name: name.into(),
            image: image.into(),
            isolated: false,
            timeout_seconds: None,
        }
    }

    pub fn isolated(mut self, value: bool) -> Self {
        self.isolated = value;
        self
    }

    pub fn timeout(mut self, seconds: u64) -> Self {
        self.timeout_seconds = Some(seconds);
        self
    }

    pub fn execute(self) -> Result<()> {
        let span = TestExecutionSpan::new(&self.name, &self.image);
        span.set_isolated(self.isolated);

        if let Some(timeout) = self.timeout_seconds {
            span.set_timeout(timeout);
        }

        let _guard = span.enter();

        run_test_with_config(self.name, self.image, self.isolated, self.timeout_seconds)
    }
}

// Usage
TestBuilder::new("my_test", "alpine:latest")
    .isolated(true)
    .timeout(30)
    .execute()?;
```

### Pattern 4: Metric Aggregation

```rust
#[cfg(feature = "otel-metrics")]
use clnrm_core::telemetry::generated::metrics::{
    TestExecutionMetric,
    ContainerOperationMetric,
};
use opentelemetry::global;
use std::time::Instant;

pub struct TelemetryCollector {
    meter: opentelemetry::metrics::Meter,
}

impl TelemetryCollector {
    pub fn new() -> Self {
        Self {
            meter: global::meter("clnrm"),
        }
    }

    pub fn record_test(&self, name: &str, duration_ms: f64, result: &str) {
        #[cfg(feature = "otel-metrics")]
        {
            let metric = TestExecutionMetric::new(&self.meter);
            metric.record(duration_ms, name, result);
        }
    }

    pub fn record_container_op(&self, image: &str, operation: &str, duration_ms: f64) {
        #[cfg(feature = "otel-metrics")]
        {
            let metric = ContainerOperationMetric::new(&self.meter);
            metric.record(duration_ms, image, operation);
        }
    }
}
```

### Pattern 5: Error Handling with Spans

```rust
use clnrm_core::telemetry::generated::spans::TestExecutionSpan;
use tracing::error;

pub fn execute_test_with_error_handling(name: &str, image: &str) -> Result<()> {
    let span = TestExecutionSpan::new(name, image);
    let _guard = span.enter();

    match run_test_steps() {
        Ok(()) => {
            span.set_result("pass");
            Ok(())
        }
        Err(e) => {
            error!("Test failed: {}", e);
            span.set_result("fail");
            span.set_error_message(&e.to_string());
            Err(e)
        }
    }
}
```

## Type Safety Examples

### Compile-Time Enforcement

```rust
// ✅ COMPILES - All required attributes provided
let span = TestExecutionSpan::new("test", "alpine:latest");

// ❌ WON'T COMPILE - Missing required attribute
let span = TestExecutionSpan::new("test"); // Error: missing image parameter

// ❌ WON'T COMPILE - Wrong type
let span = TestExecutionSpan::new(123, "alpine:latest"); // Error: expected &str
```
