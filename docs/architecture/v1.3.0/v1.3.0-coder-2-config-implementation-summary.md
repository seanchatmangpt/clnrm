## Validation Rules

### Port Validation
```rust
// Range: 0 or >= 1024
if port > 0 && port < 1024 {
    return Err("Port must be >= 1024 or 0 for auto-discovery");
}

// Conflict: otlp_port ≠ admin_port
if otlp_port == admin_port && otlp_port > 0 {
    return Err("OTLP port and admin port must be different");
}
```

### Coverage Validation
```rust
// Range: 0.0-100.0
if !(0.0..=100.0).contains(&coverage_threshold) {
    return Err("Coverage threshold must be between 0.0 and 100.0");
}
```

### 80/20 Mode Validation
```rust
// Requires critical_spans and required_attributes
if mode == ValidationMode::EightyTwenty && eighty_twenty.is_none() {
    return Err("80/20 validation mode requires [weaver.80_20] section");
}

if enabled && critical_spans.is_empty() {
    return Err("80/20 mode requires at least one critical span");
}
```

### Collector Validation
```rust
// Endpoint required if use_existing = true
if use_existing && endpoint.is_none() {
    return Err("Collector endpoint must be provided when use_existing = true");
}

// Endpoint format validation
if !endpoint.starts_with("http://") && !endpoint.starts_with("https://") {
    return Err("Collector endpoint must start with http:// or https://");
}
```

[weaver]
enabled = true
registry_path = "custom/registry/"

fail_fast = true

[weaver.validation]
mode = "strict"

[weaver.80_20]
enabled = false

[weaver.collector]

[weaver.reports]
json_report = true
html_report = true
junit_report = true

[weaver.performance]
buffer_size = 2097152

[[steps]]
