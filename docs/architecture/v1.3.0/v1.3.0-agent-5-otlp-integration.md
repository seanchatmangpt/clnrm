**Status:** Design Specification

## Executive Summary

This document specifies the integration architecture for streaming OpenTelemetry Protocol (OTLP) telemetry from clnrm test execution to Weaver live-check validation, establishing Weaver as the **single source of truth** for feature validation.

**Key Principle:** Tests can lie. Schemas don't. Weaver validation proves features work.

---

## Architecture Overview

### High-Level Flow

```

│                                                                 │

│                                                      │         │

                                           OTLP Stream │ (HTTP/gRPC)

┌──────────────────────────────────────────────────────▼─────────┐
│ Weaver Live-Check Validation                                   │
│                                                                 │
│  ┌──────────────┐      ┌──────────────┐      ┌──────────────┐ │
│  │OTLP Receiver │─────▶│Schema Matcher│─────▶│Conformance   │ │
```

### Phase 1: Dynamic OTLP Configuration

**Goal:** Configure OTLP exporter to target Weaver's actual runtime port.

#### 1.1 Port Discovery

```rust
// crates/clnrm-core/src/telemetry/weaver_controller.rs

impl WeaverController {
    /// Start Weaver and discover actual OTLP port
    pub async fn start_with_port_discovery(&self) -> Result<u16> {
        // Start Weaver process
        let process = self.start_weaver_process().await?;

        // Wait for HTTP server ready
        self.wait_for_health_check(process.port).await?;

        // Discover OTLP receiver port (may differ from HTTP port)
        let otlp_port = self.discover_otlp_port(process.port).await?;

        tracing::info!(
            http_port = process.port,
            otlp_port = otlp_port,
            "Weaver started with discovered ports"
        );

        Ok(otlp_port)
    }

    async fn discover_otlp_port(&self, http_port: u16) -> Result<u16> {
        // Query Weaver's /config or /metrics endpoint for OTLP port
        let client = reqwest::Client::new();
        let url = format!("http://localhost:{}/config", http_port);

        let response: WeaverConfig = client.get(&url)
            .timeout(Duration::from_secs(5))
            .send()
            .await
            .map_err(|e| CleanroomError::weaver_error(
                format!("Failed to discover OTLP port: {}", e)
            ))?
            .json()
            .await?;

        Ok(response.otlp_receiver_port)
    }
}

#[derive(Deserialize)]
struct WeaverConfig {
    otlp_receiver_port: u16,
    http_port: u16,
    registry_path: PathBuf,
}
```

### Phase 2: Stream Management

**Goal:** Ensure reliable telemetry delivery with proper lifecycle management.

#### 2.1 Span Flush Guarantees

```rust
// crates/clnrm-core/src/telemetry/mod.rs

pub struct OtelGuard {
    provider: TracerProvider,
    metrics_controller: Option<PushController>,
    flush_timeout: Duration,
}

impl OtelGuard {
    /// Ensure all spans flushed before shutdown
    pub fn flush_all(&self) -> Result<()> {
        tracing::info!("Flushing all telemetry before shutdown");

        // Flush traces
        for result in self.provider.force_flush() {
            result.map_err(|e| CleanroomError::telemetry_error(
                format!("Trace flush failed: {}", e)
            ))?;
        }

        // Flush metrics
        if let Some(controller) = &self.metrics_controller {
            controller.stop()
                .map_err(|e| CleanroomError::telemetry_error(
                    format!("Metrics flush failed: {}", e)
                ))?;
        }

        tracing::info!("All telemetry flushed successfully");
        Ok(())
    }
}

impl Drop for OtelGuard {
    fn drop(&mut self) {
        // Attempt graceful flush with timeout
        let start = Instant::now();

        match self.flush_all() {
            Ok(_) => {
                let elapsed = start.elapsed();
                tracing::debug!(?elapsed, "Telemetry flushed on shutdown");
            }
            Err(e) => {
                tracing::warn!(error = %e, "Failed to flush telemetry on shutdown");
            }
        }

        // Shutdown provider (this is final)
        let _ = self.provider.shutdown();
    }
}
```

#### 2.2 Backpressure Handling

```rust
#[derive(Clone, Debug)]
pub struct RetryConfig {
    pub max_retries: u32,
    pub initial_backoff: Duration,
    pub max_backoff: Duration,
    pub backoff_multiplier: f64,
}

impl Default for RetryConfig {
    fn default() -> Self {
        Self {
            max_retries: 5,
            initial_backoff: Duration::from_millis(100),
            max_backoff: Duration::from_secs(30),
            backoff_multiplier: 2.0,
        }
    }
}

impl RetryConfig {
    /// Calculate backoff for attempt N
    pub fn backoff_for_attempt(&self, attempt: u32) -> Duration {
        let backoff_ms = (self.initial_backoff.as_millis() as f64)
            * self.backoff_multiplier.powi(attempt as i32);

        Duration::from_millis(backoff_ms as u64).min(self.max_backoff)
    }
}

// Integration with opentelemetry_otlp::ExportConfig
fn configure_exporter_retry(config: &RetryConfig) -> opentelemetry_otlp::ExportConfig {
    opentelemetry_otlp::ExportConfig {
        timeout: config.max_backoff,
        // Note: opentelemetry_otlp handles retries internally
        // We configure our timeout based on max_backoff to allow for retries
    }
}
```

#### 2.3 Reconnection Logic

```rust
// crates/clnrm-core/src/telemetry/stream_manager.rs

pub struct OtlpStreamManager {
    endpoint: String,
    retry_config: RetryConfig,
    health_check_interval: Duration,
}

impl OtlpStreamManager {
    pub async fn ensure_connection(&self) -> Result<()> {
        let mut attempt = 0;

        loop {
            match self.check_weaver_health().await {
                Ok(_) => {
                    tracing::info!("Weaver connection established");
                    return Ok(());
                }
                Err(e) if attempt < self.retry_config.max_retries => {
                    let backoff = self.retry_config.backoff_for_attempt(attempt);
                    tracing::warn!(
                        attempt,
                        ?backoff,
                        error = %e,
                        "Weaver connection failed, retrying"
                    );

                    tokio::time::sleep(backoff).await;
                    attempt += 1;
                }
                Err(e) => {
                    return Err(CleanroomError::weaver_error(
                        format!("Max retries exceeded: {}", e)
                    ));
                }
            }
        }
    }

    async fn check_weaver_health(&self) -> Result<()> {
        let client = reqwest::Client::new();
        let http_endpoint = self.derive_http_endpoint();

        client.get(&format!("{}/health", http_endpoint))
            .timeout(Duration::from_secs(5))
            .send()
            .await
            .map_err(|e| CleanroomError::weaver_error(
                format!("Health check failed: {}", e)
            ))?;

        Ok(())
    }

    fn derive_http_endpoint(&self) -> String {
        // Convert OTLP endpoint to HTTP control endpoint
        // Example: http://localhost:4318 -> http://localhost:8080
        self.endpoint.replace(":4318", ":8080")
    }
}
```

### Phase 3: Stop Coordination

**Goal:** Gracefully shutdown Weaver and retrieve conformance reports.

#### 3.1 Stop Signal Handling

```rust
// crates/clnrm-core/src/telemetry/weaver_controller.rs

impl WeaverController {
    /// Gracefully stop Weaver and collect conformance report
    pub async fn stop_with_report(&self) -> Result<ConformanceReport> {
        tracing::info!("Initiating Weaver graceful shutdown");

        // Step 1: Flush all pending telemetry
        self.flush_telemetry().await?;

        // Step 2: Send stop signal via HTTP
        let stop_result = self.send_stop_signal().await?;

        // Step 3: Wait for report generation
        let report_path = self.wait_for_report_generation(
            Duration::from_secs(30)
        ).await?;

        // Step 4: Parse conformance report
        let report = ConformanceReport::load_from_file(&report_path)?;

        // Step 5: Kill process if still running
        self.ensure_process_stopped().await?;

        Ok(report)
    }

    async fn send_stop_signal(&self) -> Result<StopResponse> {
        let client = reqwest::Client::new();
        let url = format!("http://localhost:{}/stop", self.http_port);

        let response: StopResponse = client.get(&url)
            .timeout(Duration::from_secs(10))
            .send()
            .await
            .map_err(|e| CleanroomError::weaver_error(
                format!("Failed to send stop signal: {}", e)
            ))?
            .json()
            .await?;

        tracing::info!(
            report_path = %response.report_path,
            "Weaver acknowledged stop signal"
        );

        Ok(response)
    }

    async fn wait_for_report_generation(&self, timeout: Duration) -> Result<PathBuf> {
        let start = Instant::now();
        let expected_path = self.report_output_path();

        while start.elapsed() < timeout {
            if expected_path.exists() {
                // Wait a bit more to ensure file is fully written
                tokio::time::sleep(Duration::from_millis(500)).await;
                return Ok(expected_path);
            }

            tokio::time::sleep(Duration::from_millis(100)).await;
        }

        Err(CleanroomError::weaver_error(
            format!("Report generation timeout after {:?}", timeout)
        ))
    }

    async fn ensure_process_stopped(&self) -> Result<()> {
        if let Some(mut child) = self.process.lock().await.take() {
            // Try graceful shutdown first
            match tokio::time::timeout(
                Duration::from_secs(5),
                child.wait()
            ).await {
                Ok(Ok(status)) => {
                    tracing::info!(?status, "Weaver process exited");
                }
                Ok(Err(e)) => {
                    tracing::warn!(error = %e, "Failed to wait for process");
                }
                Err(_) => {
                    // Timeout - force kill
                    tracing::warn!("Weaver process did not exit gracefully, killing");
                    child.kill().await?;
                }
            }
        }

        Ok(())
    }
}

#[derive(Deserialize, Debug)]
struct StopResponse {
    status: String,
    report_path: PathBuf,
    validation_count: usize,
    error_count: usize,
}
```

#### 3.2 Conformance Report Parsing

```rust
// crates/clnrm-core/src/telemetry/validation_analyzer.rs

#[derive(Debug, Clone)]
pub struct ConformanceReport {
    pub registry_path: PathBuf,
    pub telemetry_received: usize,
    pub validations: Vec<ValidationResult>,
    pub errors: Vec<ConformanceError>,
    pub summary: ConformanceSummary,
}

#[derive(Debug, Clone)]
pub struct ValidationResult {
    pub signal_type: SignalType,  // Trace, Metric, Log
    pub schema_id: String,        // e.g., "clnrm.test.execution"
    pub matched: bool,
    pub violations: Vec<String>,
}

#[derive(Debug, Clone)]
pub struct ConformanceSummary {
    pub total_signals: usize,
    pub validated_signals: usize,
    pub failed_signals: usize,
    pub conformance_rate: f64,
}

impl ConformanceReport {
    pub fn load_from_file(path: &Path) -> Result<Self> {
        let content = std::fs::read_to_string(path)
            .map_err(|e| CleanroomError::weaver_error(
                format!("Failed to read conformance report: {}", e)
            ))?;

        // Parse Weaver's JSON output format
        let report: WeaverLiveCheckOutput = serde_json::from_str(&content)
            .map_err(|e| CleanroomError::weaver_error(
                format!("Failed to parse conformance report: {}", e)
            ))?;

        Ok(Self::from_weaver_output(report))
    }

    fn from_weaver_output(output: WeaverLiveCheckOutput) -> Self {
        let summary = ConformanceSummary {
            total_signals: output.signals_received,
            validated_signals: output.signals_validated,
            failed_signals: output.signals_failed,
            conformance_rate: output.signals_validated as f64
                / output.signals_received as f64,
        };

        Self {
            registry_path: output.registry_path,
            telemetry_received: output.signals_received,
            validations: output.validations.into_iter()
                .map(ValidationResult::from)
                .collect(),
            errors: output.errors,
            summary,
        }
    }

    /// Check if validation passed
    pub fn is_valid(&self) -> bool {
        self.errors.is_empty() && self.summary.conformance_rate >= 1.0
    }

    /// Generate human-readable summary
    pub fn format_summary(&self) -> String {
        format!(
            "Weaver Validation Results:\n\
             • Signals Received: {}\n\
             • Signals Validated: {}\n\
             • Conformance Rate: {:.1}%\n\
             • Status: {}\n",
            self.summary.total_signals,
            self.summary.validated_signals,
            self.summary.conformance_rate * 100.0,
            if self.is_valid() { "PASS ✅" } else { "FAIL ❌" }
        )
    }
}

#[derive(Deserialize)]
struct WeaverLiveCheckOutput {
    registry_path: PathBuf,
    signals_received: usize,
    signals_validated: usize,
    signals_failed: usize,
    validations: Vec<WeaverValidation>,
    errors: Vec<ConformanceError>,
}
```

#### 4.1 Signal Handling

```rust
// crates/clnrm-core/src/telemetry/weaver_controller.rs

use tokio::signal;

pub struct WeaverController {
    // ... existing fields ...
    shutdown_tx: broadcast::Sender<ShutdownReason>,
}

#[derive(Debug, Clone)]
pub enum ShutdownReason {
    SigInt,           // Ctrl+C
    SigHup,           // Configuration reload
    HttpStop,         // Programmatic /stop
    InactivityTimeout, // No telemetry for N seconds
}

impl WeaverController {
    pub async fn run_with_signal_handling(&self) -> Result<ConformanceReport> {
        let mut shutdown_rx = self.shutdown_tx.subscribe();

        // Spawn signal handlers
        let shutdown_tx_clone = self.shutdown_tx.clone();
        tokio::spawn(async move {
            if let Ok(()) = signal::ctrl_c().await {
                let _ = shutdown_tx_clone.send(ShutdownReason::SigInt);
            }
        });

        #[cfg(unix)]
        {
            let shutdown_tx_clone = self.shutdown_tx.clone();
            tokio::spawn(async move {
                let mut sighup = signal::unix::signal(
                    signal::unix::SignalKind::hangup()
                ).expect("Failed to setup SIGHUP handler");

                if sighup.recv().await.is_some() {
                    let _ = shutdown_tx_clone.send(ShutdownReason::SigHup);
                }
            });
        }

        // Wait for shutdown signal
        let reason = shutdown_rx.recv().await
            .map_err(|e| CleanroomError::weaver_error(
                format!("Shutdown channel error: {}", e)
            ))?;

        tracing::info!(?reason, "Shutdown signal received");

        // Graceful shutdown with report
        self.stop_with_report().await
    }
}
```

#### 4.2 Inactivity Timeout

```rust
pub struct InactivityMonitor {
    last_telemetry: Arc<RwLock<Instant>>,
    timeout: Duration,
    shutdown_tx: broadcast::Sender<ShutdownReason>,
}

impl InactivityMonitor {
    pub fn new(
        timeout: Duration,
        shutdown_tx: broadcast::Sender<ShutdownReason>,
    ) -> Self {
        Self {
            last_telemetry: Arc::new(RwLock::new(Instant::now())),
            timeout,
            shutdown_tx,
        }
    }

    pub fn record_activity(&self) {
        let mut last = self.last_telemetry.write().unwrap();
        *last = Instant::now();
    }

    pub async fn monitor_loop(self: Arc<Self>) {
        let check_interval = self.timeout / 4;

        loop {
            tokio::time::sleep(check_interval).await;

            let last = *self.last_telemetry.read().unwrap();
            let elapsed = last.elapsed();

            if elapsed > self.timeout {
                tracing::warn!(
                    ?elapsed,
                    timeout = ?self.timeout,
                    "Inactivity timeout reached"
                );

                let _ = self.shutdown_tx.send(ShutdownReason::InactivityTimeout);
                break;
            }
        }
    }
}

// Integration with OTLP exporter
impl OtlpStreamManager {
    pub fn with_inactivity_monitor(
        mut self,
        monitor: Arc<InactivityMonitor>,
    ) -> Self {
        // Hook into exporter to record activity
        // This requires custom exporter wrapper
        self.activity_monitor = Some(monitor);
        self
    }
}
```

## Error Handling

### Common Failure Modes

| Failure | Detection | Recovery |
|---------|----------|----------|
| **Weaver not installed** | Binary check on startup | Error with install instructions |
| **Port already in use** | Health check failure | Try alternate port or fail fast |
| **OTLP export timeout** | Timeout on send | Retry with backoff, log warning |
| **Schema validation failure** | Conformance report errors | Fail test, report violations |
| **Weaver crash** | Process exit detection | Attempt restart once, then fail |
| **Report generation timeout** | File polling timeout | Warning + partial results |

### Error Messages

```rust
pub enum WeaverError {
    BinaryNotFound {
        expected_path: PathBuf,
        install_url: String,
    },
    PortUnavailable {
        port: u16,
        reason: String,
    },
    OtlpExportFailed {
        attempt: u32,
        max_retries: u32,
        last_error: String,
    },
    SchemaViolation {
        signal_type: String,
        expected_schema: String,
        violations: Vec<String>,
    },
    ProcessCrashed {
        exit_code: Option<i32>,
        stderr: String,
    },
}

impl Display for WeaverError {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        match self {
            Self::BinaryNotFound { expected_path, install_url } => write!(
                f,
                "Weaver binary not found at {:?}\n\
                 Install from: {}",
                expected_path, install_url
            ),
            Self::PortUnavailable { port, reason } => write!(
                f,
                "Port {} unavailable: {}",
                port, reason
            ),
            Self::OtlpExportFailed { attempt, max_retries, last_error } => write!(
                f,
                "OTLP export failed after {}/{} retries: {}",
                attempt, max_retries, last_error
            ),
            Self::SchemaViolation { signal_type, expected_schema, violations } => write!(
                f,
                "Schema violation for {} (expected schema: {})\n\
                 Violations:\n{}",
                signal_type,
                expected_schema,
                violations.join("\n")
            ),
            Self::ProcessCrashed { exit_code, stderr } => write!(
                f,
                "Weaver process crashed (exit code: {:?})\nStderr:\n{}",
                exit_code, stderr
            ),
        }
    }
}
```

## Performance Considerations

### Telemetry Overhead

**Expected Impact:**
- **Span creation:** ~5-10µs per span
- **OTLP export (batch):** ~10-50ms per batch (1000 spans)
- **Weaver validation:** ~1-5ms per signal
- **Total overhead:** <5% for typical test suites

**Mitigation Strategies:**
1. **Batching:** Export spans in batches of 512-1024
2. **Async export:** Never block test execution on OTLP send
3. **Sampling:** Use 100% for validation, <10% for production
4. **Buffering:** 10MB buffer for burst telemetry

## Security Considerations

### Port Exposure

**Risk:** Weaver OTLP receiver exposed on localhost.

**Mitigation:**
- Bind to 127.0.0.1 only (no 0.0.0.0)
- Random port selection (avoid predictable ports)
- Firewall rules (if deployed on remote host)

### Telemetry Data Leakage

**Risk:** Sensitive data in span attributes/logs.

**Mitigation:**
- Redaction filters for known sensitive fields
- Schema enforcement (no arbitrary attributes)
- Encryption for OTLP transport (if over network)

```rust
pub struct TelemetryRedactor {
    sensitive_patterns: Vec<Regex>,
}

impl TelemetryRedactor {
    pub fn redact_span_attributes(&self, attrs: &mut HashMap<String, String>) {
        for (key, value) in attrs.iter_mut() {
            if self.is_sensitive(key) {
                *value = "[REDACTED]".to_string();
            } else {
                for pattern in &self.sensitive_patterns {
                    if pattern.is_match(value) {
                        *value = pattern.replace_all(value, "[REDACTED]").to_string();
                    }
                }
            }
        }
    }

    fn is_sensitive(&self, key: &str) -> bool {
        matches!(
            key.to_lowercase().as_str(),
            "password" | "token" | "api_key" | "secret" | "credential"
        )
    }
}
```

```rust
    pub otlp_exports_total: Counter,
    pub otlp_export_failures: Counter,
    pub weaver_validations_total: Counter,
    pub weaver_validation_failures: Counter,
    pub conformance_report_latency: Histogram,
    pub telemetry_overhead_ms: Histogram,
}

impl WeaverMetrics {
    pub fn record_export_success(&self) {
        self.otlp_exports_total.inc();
    }

    pub fn record_export_failure(&self) {
        self.otlp_export_failures.inc();
    }

    pub fn export_success_rate(&self) -> f64 {
        let total = self.otlp_exports_total.get();
        let failures = self.otlp_export_failures.get();

        if total == 0 {
            return 1.0;
        }

        (total - failures) as f64 / total as f64
    }
```

```rust
// crates/clnrm-core/src/cli/commands/run.rs

pub async fn execute_run_command(
    test_path: PathBuf,
    weaver_validate: bool,
    weaver_registry: Option<PathBuf>,
) -> Result<()> {
    if weaver_validate {
        run_with_weaver_validation(test_path, weaver_registry).await
    } else {
        run_without_validation(test_path).await
    }
}

async fn run_with_weaver_validation(
    test_path: PathBuf,
    registry_path: Option<PathBuf>,
) -> Result<()> {
    // Discover registry path
    let registry = registry_path.unwrap_or_else(|| {
        PathBuf::from("./registry")
    });

    println!("🔍 Starting Weaver validation...");

    // Step 1: Start Weaver
    let weaver = WeaverController::new(&registry)?;
    let otlp_port = weaver.start_with_port_discovery().await?;
    println!("✅ Weaver started on port {}", otlp_port);

    // Step 2: Configure OTEL
    let otel_config = OtelConfig::for_weaver(otlp_port);
    let _otel_guard = init_otel(otel_config)?;
    println!("✅ OTLP exporter configured");

    // Step 3: Execute test
    println!("🚀 Running test: {}", test_path.display());
    let test_result = execute_test(&test_path).await?;

    if test_result.passed() {
        println!("✅ Test execution: PASS");
    } else {
        println!("❌ Test execution: FAIL");
    }

    // Step 4: Get conformance report
    println!("📊 Generating conformance report...");
    let conformance = weaver.stop_with_report().await?;

    // Step 5: Display results
    println!("\n{}", conformance.format_summary());

    if conformance.is_valid() {
        println!("\n✅ WEAVER VALIDATION: PASS");
        println!("Feature is production-ready! 🎉");
    } else {
        println!("\n❌ WEAVER VALIDATION: FAIL");
        println!("Schema violations detected:");
        for error in &conformance.errors {
            println!("  • {}", error);
        }
        return Err(CleanroomError::validation_failed(
            "Weaver validation failed"
        ));
    }

    Ok(())
}
```
