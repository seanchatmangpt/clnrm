## 5. Feature Detection

### 5.1 Runtime Detection (Code-Level)

**Rust Implementation:**

```rust
// crates/clnrm-core/src/config.rs

use serde::{Deserialize, Serialize};

#[derive(Debug, Clone, Deserialize, Serialize)]
pub struct TestConfig {
    pub metadata: TestMetadata,
    pub services: HashMap<String, ServiceConfig>,
    pub steps: Vec<StepConfig>,
    pub assertions: Option<Assertions>,

    // NEW in v1.3.0 (OPTIONAL)
    #[serde(default)]
    pub live_check: Option<LiveCheckConfig>,
}

#[derive(Debug, Clone, Deserialize, Serialize)]
pub struct LiveCheckConfig {
    pub enabled: bool,
    pub registry_path: String,
    pub validation_mode: ValidationMode,
    pub fail_on_violation: bool,
    pub max_retries: Option<u32>,
    pub timeout_ms: Option<u64>,
    pub expected_spans: Option<Vec<String>>,
    pub expected_metrics: Option<Vec<String>>,
    pub required_attributes: Option<Vec<String>>,
}

// Feature detection
pub fn supports_live_check(config: &TestConfig) -> bool {
    config.live_check
        .as_ref()
        .map(|lc| lc.enabled)
        .unwrap_or(false)
}

// Execution flow
pub async fn execute_test(config: &TestConfig) -> Result<TestResult> {
    if supports_live_check(config) {
        // v1.3.0 flow: Weaver live-check validation
        execute_with_live_check(config).await
    } else {
        // v1.2.1 flow: Legacy execution (no Weaver)
        execute_legacy_flow(config).await
    }
}

// Legacy flow (v1.2.1 compatible)
async fn execute_legacy_flow(config: &TestConfig) -> Result<TestResult> {
    let env = CleanroomEnvironment::new().await?;

    // Start services
    for (name, svc_config) in &config.services {
        env.start_service(name, svc_config).await?;
    }

    // Execute steps
    let mut step_results = Vec::new();
    for step in &config.steps {
        let result = env.execute_step(step).await?;
        step_results.push(result);
    }

    // Validate assertions
    validate_assertions(&config.assertions, &step_results)?;

    Ok(TestResult {
        passed: true,
        duration_ms: 450,
        used_live_check: false,  // ← NO Weaver
        conformance: None,
    })
}

// Live-check flow (v1.3.0)
async fn execute_with_live_check(config: &TestConfig) -> Result<TestResult> {
    let env = CleanroomEnvironment::new().await?;

    // Extract live-check config
    let lc_config = config.live_check.as_ref()
        .ok_or_else(|| CleanroomError::internal_error("live_check missing"))?;

    // Initialize Weaver controller
    let weaver = WeaverController::new(&lc_config.registry_path)?;

    // Start services
    for (name, svc_config) in &config.services {
        env.start_service(name, svc_config).await?;
    }

    // Execute steps
    let mut step_results = Vec::new();
    for step in &config.steps {
        let result = env.execute_step(step).await?;
        step_results.push(result);
    }

    // Validate assertions
    validate_assertions(&config.assertions, &step_results)?;

    // ← NEW: Weaver live-check validation
    let conformance = weaver.validate_conformance(lc_config).await?;

    // Handle conformance result
    let passed = match (conformance.passed, lc_config.fail_on_violation) {
        (true, _) => true,           // Conformance passed
        (false, true) => false,      // Conformance failed, fail_on_violation=true
        (false, false) => true,      // Conformance failed, fail_on_violation=false (warn only)
    };

    Ok(TestResult {
        passed,
        duration_ms: 1230,
        used_live_check: true,       // ← YES Weaver
        conformance: Some(conformance),
    })
}
```

// Validate TOML schema version
pub fn validate_config_schema(config: &TestConfig) -> Result<SchemaVersion> {
    if config.live_check.is_some() {

        Ok(SchemaVersion::V1_3_0)
    } else {
        // v1.2.1 schema detected
        Ok(SchemaVersion::V1_2_1)

// Warn about mixed versions
pub fn warn_mixed_versions(configs: &[TestConfig]) {
    let v1_2_1_count = configs.iter()
        .filter(|c| c.live_check.is_none())
        .count();
    let v1_3_0_count = configs.len() - v1_2_1_count;

    if v1_2_1_count > 0 && v1_3_0_count > 0 {
        eprintln!(
            "Warning: Mixed TOML versions detected:\n\
             - {} tests using v1.2.1 schema (no [test.live_check])\n\
             - {} tests using v1.3.0 schema (with [test.live_check])\n\
             \n\
             This is supported but consider migrating all tests to v1.3.0.\n",
            v1_2_1_count,
            v1_3_0_count
        );
    }
}

pub fn validate_live_check_config(lc: &LiveCheckConfig) -> Result<()> {
    // Validate registry_path exists
    if !Path::new(&lc.registry_path).exists() {
        return Err(CleanroomError::config_error(
            format!("Registry path not found: {}", lc.registry_path)
        ));
    }

    // Validate validation_mode is recognized
    match lc.validation_mode {
        ValidationMode::Strict | ValidationMode::EightyTwenty | ValidationMode::Permissive => {},
        _ => return Err(CleanroomError::config_error(
            "Invalid validation_mode (expected: strict, 80_20, permissive)"
        )),
    }

    // Validate timeout_ms is reasonable
    if let Some(timeout) = lc.timeout_ms {
        if timeout < 100 || timeout > 60_000 {
            return Err(CleanroomError::config_error(
                "timeout_ms must be between 100 and 60000"
            ));
        }
    }

    Ok(())
}
